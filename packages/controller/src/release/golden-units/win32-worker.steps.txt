Run these in an ELEVATED PowerShell (Run as administrator): registering a boot-triggered task for another account's S4U logon needs an administrator, and the unit itself asks for no more than LeastPrivilege. Nothing below has been run by this program. Run native\install\Install-NimbleNewtLauncher.ps1 first (it makes the account, the launcher and worker.env).
if (-not ((Test-Path -LiteralPath 'C:\Program Files\nodejs\node.exe') -and (Test-Path -LiteralPath 'C:\Program Files\NimbleNewt\release\scripts\remote-worker.mjs') -and (Test-Path -LiteralPath 'C:\ProgramData\NimbleNewt\worker.json') -and (Test-Path -LiteralPath 'C:\ProgramData\NimbleNewt\worker.env') -and (Test-Path -LiteralPath 'C:\Program Files\NimbleNewt\release'))) { throw 'node, worker entry, worker configuration, worker.env or working directory is missing' }
$xml = @'
<?xml version="1.0" encoding="UTF-16"?>
<!-- Generated by nimblenewt-service.mjs (print-worker). No secret and no password is in it: the account has none stored (S4U), the environment is the file named in Arguments (Node's env-file option). Registered by an administrator, never by this program. -->
<Task version="1.3" xmlns="http://schemas.microsoft.com/windows/2004/02/mit/task">
  <RegistrationInfo>
    <Description>NimbleNewt worker (NimbleNewtWorker), runs as BOX\nimblenewt-worker without a stored password (S4U)</Description>
  </RegistrationInfo>
  <Triggers>
    <BootTrigger>
      <Enabled>true</Enabled>
      <Delay>PT1M</Delay>
    </BootTrigger>
  </Triggers>
  <Settings>
    <AllowStartOnDemand>true</AllowStartOnDemand>
    <RestartOnFailure><Interval>PT1M</Interval><Count>3</Count></RestartOnFailure>
    <MultipleInstancesPolicy>IgnoreNew</MultipleInstancesPolicy>
    <DisallowStartIfOnBatteries>false</DisallowStartIfOnBatteries>
    <StopIfGoingOnBatteries>false</StopIfGoingOnBatteries>
    <AllowHardTerminate>true</AllowHardTerminate>
    <StartWhenAvailable>true</StartWhenAvailable>
    <RunOnlyIfNetworkAvailable>false</RunOnlyIfNetworkAvailable>
    <WakeToRun>false</WakeToRun>
    <Enabled>true</Enabled>
    <Hidden>false</Hidden>
    <ExecutionTimeLimit>PT0S</ExecutionTimeLimit>
    <Priority>7</Priority>
  </Settings>
  <Principals>
    <Principal id="Worker">
      <UserId>BOX\nimblenewt-worker</UserId>
      <LogonType>S4U</LogonType>
      <RunLevel>LeastPrivilege</RunLevel>
    </Principal>
  </Principals>
  <Actions Context="Worker">
    <Exec>
      <Command>C:\Program Files\nodejs\node.exe</Command>
      <Arguments>&quot;--env-file=C:\ProgramData\NimbleNewt\worker.env&quot; &quot;C:\Program Files\NimbleNewt\release\scripts\remote-worker.mjs&quot; &quot;--config&quot; &quot;C:\ProgramData\NimbleNewt\worker.json&quot;</Arguments>
      <WorkingDirectory>C:\Program Files\NimbleNewt\release</WorkingDirectory>
    </Exec>
  </Actions>
</Task>
'@
Register-ScheduledTask -TaskName 'NimbleNewtWorker' -Xml $xml   # the account is the S4U principal in the XML: add no user and no password option
Start-ScheduledTask -TaskName 'NimbleNewtWorker'
Get-ScheduledTaskInfo -TaskName 'NimbleNewtWorker'
.\native\install\Install-NimbleNewtLauncher.ps1 -Action Verify   # also reports the task's principal and trigger
